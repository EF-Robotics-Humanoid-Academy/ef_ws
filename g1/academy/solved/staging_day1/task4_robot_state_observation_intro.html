§2Knowledge — defensive reads and map validation

Defensive attribute access: SportModeState_ field names vary a little across SDK/firmware versions (position vs. pos vs. position_w). A small _first_attr(obj, names, default) helper tries each candidate name in order instead of assuming one exact spelling — the same defensive pattern sdk_wrapper.py uses internally (see _read()).

Validate before you trust: for the occupancy-grid reader, check width, height, resolution, and that len(data) actually matches width*height before reshaping — never reshape blindly and let numpy silently produce garbage on a short/misaligned buffer.

class Latest:
    def __init__(self, topic, message_type, queue_len=10):
        self.message = None
        self.timestamp = 0.0
        self.subscriber = ChannelSubscriber(topic, message_type)
        self.subscriber.Init(self._callback, queue_len)

    def _callback(self, message):
        self.message = message
        self.timestamp = time.time()

    def fresh(self, max_age_s=0.5):
        return self.message is not None and time.time() - self.timestamp <= max_age_s

def wait_for_latest(latest, timeout_s=3.0, max_age_s=1.0):
    deadline = time.time() + timeout_s
    while time.time() < deadline:
        if latest.fresh(max_age_s):
            return latest.message
        time.sleep(0.05)
    return None

def show(data):

def get_occupancygrid(timeout_s=3.0):
    msg = wait_for_latest(occupancygrid_sub, timeout_s=timeout_s)
    if msg is None:
        return None
    width, height = int(msg.width), int(msg.height)
    resolution = float(getattr(msg, "resolution", 0.0))
    data = list(msg.data)
    if width <= 0 or height <= 0 or resolution <= 0.0 or len(data) != width * height:
        raise ValueError(f"HeightMap_ shape mismatch: {width}x{height} vs {len(data)} cells")
    grid = np.asarray(data, dtype=np.float32).reshape(height, width)
    return {"timestamp": occupancygrid_sub.timestamp, "width": width, "height": height,
            "resolution": resolution, "grid": grid}

RGB-D frames come from rgbd_server_service (see realsense_rgbd_zmq_stream.py), a ZMQ PUB socket sending [rgb_jpeg, depth_png, depth_scale] multipart frames — not a DDS topic at all.

import os, struct

def get_rgbd(endpoints=None):
    import zmq
    endpoints = endpoints or ["tcp://127.0.0.1:5555", "tcp://localhost:5555"]
    last_error = None
    for endpoint in endpoints:
        ctx = zmq.Context.instance()
        sock = ctx.socket(zmq.SUB)
        sock.setsockopt(zmq.SUBSCRIBE, b"")
        sock.setsockopt(zmq.RCVTIMEO, 3000)
        sock.connect(endpoint)
        try:
            parts = sock.recv_multipart()
            if len(parts) < 3:
                continue
            scale = None
            if parts[2] != b"0" and len(parts[2]) == 4:
                scale = float(struct.unpack("f", parts[2])[0])
            return {"timestamp": time.time(), "endpoint": endpoint, "rgb_jpeg": bytes(parts[0]),
                    "depth_png": None if parts[1] == b"0" else bytes(parts[1]), "depth_scale": scale}
        except Exception as exc:
            last_error = exc
        finally:
            sock.close(0)
    if last_error is not None:
        raise last_error
    return None

def show_rgbd(frame):
    if frame is None:
        print("no RGB-D frame yet"); return
    import cv2, numpy as np
    rgb = cv2.imdecode(np.frombuffer(frame["rgb_jpeg"], dtype=np.uint8), cv2.IMREAD_COLOR)
    plt.figure(figsize=(5, 4)); plt.imshow(cv2.cvtColor(rgb, cv2.COLOR_BGR2RGB)); plt.axis("off"); plt.show()
    if frame["depth_png"]:
        depth = cv2.imdecode(np.frombuffer(frame["depth_png"], dtype=np.uint8), cv2.IMREAD_UNCHANGED)
        plt.figure(figsize=(5, 4)); plt.imshow(depth); plt.colorbar(); plt.show()

def get_services():
    code, service_states = robot_state_client.ServiceList()
    if int(code) != 0:
        raise RuntimeError(f"ServiceList failed: {code}")
    rows = []
    for state in service_states or []:
        name = str(getattr(state, "name", "")).strip()
        if name:
            rows.append({
                "name": name,
                "description": SERVICE_CATALOG.get(name, ""),
                "status": None if getattr(state, "status", None) is None else int(state.status),
                "protected": None if getattr(state, "protect", None) is None else bool(int(state.protect)),
            })
    return rows
